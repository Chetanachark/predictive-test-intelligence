frontend/src/context/shopState.js

import initialProducts from '../data/products.js'

export function createShopState() {
  return { products: initialProducts.map(product => ({ ...product })), cart: [], orders: [] }
}

export function couponDetails(value) {
  const code = value.trim().toUpperCase()
  if (!code) return { code: '', rate: 0 }
  const rates = { SAVE10: 0.1, WELCOME20: 0.2 }
  if (!Object.hasOwn(rates, code)) throw new Error('Invalid coupon code.')
  return { code, rate: rates[code] }
}

export function orderTotals(cart, coupon = '') {
  const { code, rate } = couponDetails(coupon)
  const subtotal = cart.reduce((sum, item) => sum + item.price * item.quantity, 0)
  const discount = Math.round(subtotal * rate * 100) / 100
  return { subtotal, discount, total: Math.round((subtotal - discount) * 100) / 100, coupon: code }
}

export function updateCart(state, action) {
  if (action.type === 'clear') return { ...state, cart: [] }
  if (action.type === 'remove') {
    return { ...state, cart: state.cart.filter(item => item.id !== action.id) }
  }
  const product = state.products.find(item => item.id === action.id)
  if (!product) return state
  const existing = state.cart.find(item => item.id === action.id)
  const quantity = action.type === 'add' ? (existing?.quantity ?? 0) + 1 : action.quantity
  if (!['add', 'quantity'].includes(action.type) || !Number.isInteger(quantity) ||
      quantity < 1 || quantity > Math.min(product.stock, 99)) return state
  const cart = existing
    ? state.cart.map(item => item.id === product.id ? { ...item, quantity } : item)
    : [...state.cart, { ...product, quantity }]
  return { ...state, cart }
}

// Commit stock, cart, and order history together in one React state update.
export function completeOrder(state, customer, coupon, number, date) {
  const details = Object.fromEntries(['name', 'email', 'address'].map(key => [key, String(customer[key] ?? '').trim()]))
  if (Object.values(details).some(value => !value)) throw new Error('Full name, email, and address are required.')
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(details.email)) throw new Error('Enter a valid email address.')
  if (!state.cart.length) throw new Error('Your cart is empty.')
  const totals = orderTotals(state.cart, coupon)
  const items = state.cart.map(item => {
    const product = state.products.find(product => product.id === item.id)
    if (!product || !Number.isInteger(item.quantity) || item.quantity < 1 || item.quantity > Math.min(product.stock, 99)) {
      throw new Error('The requested quantity is no longer in stock.')
    }
    return { id: product.id, name: product.name, price: product.price, quantity: item.quantity }
  })
  const order = { number, date, customer: details, items, ...totals }
  return {
    products: state.products.map(product => ({ ...product,
      stock: product.stock - (items.find(item => item.id === product.id)?.quantity ?? 0) })),
    cart: [],
    orders: [order, ...state.orders],
  }
}

frontend/src/context/StoreProvider.jsx

import { useState } from 'react'
import { StoreContext } from './store.js'
import { createShopState, updateCart, completeOrder } from './shopState.js'

export default function StoreProvider({ children }) {
  const [session, setSession] = useState(null)
  const [shop, setShop] = useState(createShopState)

  // Existing demo entry behavior; no credential verification or persistence.
  function login(email) {
    setSession({ user: { name: 'Demo Student', email: email.trim() } })
    setShop(createShopState())
  }

  function logout() {
    setSession(null)
    setShop(createShopState())
  }

  function addItem(product) {
    setShop(current => updateCart(current, { type: 'add', id: product.id }))
  }

  function setQuantity(id, quantity) {
    setShop(current => updateCart(current, { type: 'quantity', id, quantity }))
  }

  function removeItem(id) {
    setShop(current => updateCart(current, { type: 'remove', id }))
  }

  function clearCart() {
    setShop(current => updateCart(current, { type: 'clear' }))
  }

  function placeOrder(customer, coupon) {
    const number = `ORD-${crypto.randomUUID().toUpperCase()}`
    const nextShop = completeOrder(shop, customer, coupon, number, new Date().toISOString())
    setShop(nextShop)
    return nextShop.orders[0]
  }

  const total = shop.cart.reduce((sum, item) => sum + item.price * item.quantity, 0)

  return (
    <StoreContext.Provider value={{
      session, checking: false, cart: shop.cart, products: shop.products,
      orders: shop.orders, order: shop.orders[0] ?? null, total,
      login, logout, addItem, setQuantity, removeItem, clearCart, placeOrder,
    }}>
      {children}
    </StoreContext.Provider>
  )
}

frontend/src/pages/Products.jsx

import { useState } from 'react'
import { useStore } from '../context/store.js'
import ProductCard from '../components/ProductCard.jsx'

const sorters = {
  'name-asc': (a, b) => a.name.localeCompare(b.name),
  'name-desc': (a, b) => b.name.localeCompare(a.name),
  'price-asc': (a, b) => a.price - b.price || a.name.localeCompare(b.name),
  'price-desc': (a, b) => b.price - a.price || a.name.localeCompare(b.name),
}

export default function Products() {
  const { products, cart, addItem } = useStore()
  const [search, setSearch] = useState('')
  const [sort, setSort] = useState('name-asc')
  const [notice, setNotice] = useState('')
  const query = search.trim().toLowerCase()
  const visibleProducts = products.filter(product => product.name.toLowerCase().includes(query)).sort(sorters[sort])

  function handleAdd(product) {
    const quantity = cart.find(item => item.id === product.id)?.quantity ?? 0
    if (quantity >= Math.min(product.stock, 99)) {
      setNotice(`No more ${product.name} can be added.`)
      return
    }
    addItem(product)
    setNotice(`${product.name} added to cart.`)
  }

  return (
    <section>
      <p className="eyebrow">The catalog</p>
      <h1>Products</h1>
      <p>Choose your tools. Build something great.</p>
      <div className="catalog-controls">
        <label>
          Search products
          <input type="search" value={search} placeholder="Search by product name"
            onChange={event => { setSearch(event.target.value); setNotice('') }} />
        </label>
        <label>
          Sort products
          <select value={sort} onChange={event => setSort(event.target.value)}>
            <option value="name-asc">Name A-Z</option>
            <option value="name-desc">Name Z-A</option>
            <option value="price-asc">Price Low-High</option>
            <option value="price-desc">Price High-Low</option>
          </select>
        </label>
      </div>
      {visibleProducts.length ? (
        <div className="grid">
          {visibleProducts.map(product => (
            <ProductCard key={product.id} product={product} onAdd={handleAdd}
              quantityInCart={cart.find(item => item.id === product.id)?.quantity ?? 0} />
          ))}
        </div>
      ) : <p role="status">No products found.</p>}
      <p role="status">{notice}</p>
    </section>
  )
}

frontend/src/pages/Checkout.jsx

import { useRef, useState } from 'react'
import { Link, useNavigate } from 'react-router-dom'
import { money } from '../api/client.js'
import { couponDetails, orderTotals } from '../context/shopState.js'
import { useStore } from '../context/store.js'

export default function Checkout() {
  const { cart, placeOrder } = useStore()
  const [couponInput, setCouponInput] = useState('')
  const [coupon, setCoupon] = useState('')
  const [couponMessage, setCouponMessage] = useState('')
  const [couponError, setCouponError] = useState('')
  const [error, setError] = useState('')
  const [submitting, setSubmitting] = useState(false)
  const submitted = useRef(false)
  const navigate = useNavigate()
  const totals = orderTotals(cart, coupon)

  function applyCoupon() {
    setCouponError('')
    setCouponMessage('')
    try {
      const result = couponDetails(couponInput)
      setCoupon(result.code)
      setCouponMessage(result.code ? `${result.code} applied: ${result.rate * 100}% discount.` : 'No coupon applied.')
    } catch (error) {
      setCoupon('')
      setCouponError(error.message)
    }
  }

  function submit(event) {
    event.preventDefault()
    if (submitted.current || !cart.length) return
    const data = new FormData(event.currentTarget)
    const customer = Object.fromEntries(['name', 'email', 'address'].map(key => [key, data.get(key).trim()]))
    // Coupon input must be explicitly applied before placing an order.
    if (couponInput.trim().toUpperCase() !== coupon) {
      setCouponError('Apply or clear the coupon before placing your order.')
      return
    }
    setError('')
    submitted.current = true
    setSubmitting(true)
    try {
      placeOrder(customer, coupon)
      navigate('/success', { replace: true })
    } catch (error) {
      submitted.current = false
      setSubmitting(false)
      setError(error.message)
    }
  }

  return (
    <section className="panel narrow">
      <h1>Checkout</h1>
      {cart.length ? (
        <>
          <p>Subtotal: <strong>{money(totals.subtotal)}</strong></p>
          <p>Discount: <strong>{money(totals.discount)}</strong></p>
          <p>Total: <strong>{money(totals.total)}</strong></p>
          <p>Demo orders only. No payment is collected.</p>
          <form onSubmit={submit}>
            <label>Full Name<input name="name" autoComplete="name" maxLength="200" required /></label>
            <label>Email<input name="email" type="email" autoComplete="email" maxLength="254" required /></label>
            <label>Address<input name="address" autoComplete="street-address" maxLength="500" required /></label>
            <label>
              Coupon code
              <input name="coupon" value={couponInput} maxLength="30" aria-describedby="coupon-help"
                onChange={event => {
                  setCouponInput(event.target.value)
                  setCoupon('')
                  setCouponMessage('')
                  setCouponError('')
                }} />
            </label>
            <p id="coupon-help">Optional: SAVE10 for 10% off or WELCOME20 for 20% off.</p>
            <button type="button" className="secondary" onClick={applyCoupon}>Apply coupon</button>
            {couponMessage && <p role="status">{couponMessage}</p>}
            {couponError && <p role="alert" className="error">{couponError}</p>}
            {error && <p role="alert" className="error">{error}</p>}
            <button type="submit" disabled={submitting}>{submitting ? 'Placing order…' : 'Place order'}</button>
          </form>
        </>
      ) : <><p>Your cart is empty.</p><Link to="/products">Browse products</Link></>}
    </section>
  )
}

          <h2 className="order-id">Order number: {order.number}</h2>
          <p>Date: <time dateTime={order.date}>{new Date(order.date).toLocaleString('en-IN')}</time></p>
          <ul>{order.items.map(item => (
            <li key={item.id}>{item.name} × {item.quantity} — {money(item.price * item.quantity)}</li>
          ))}</ul>
          <p>Subtotal: {money(order.subtotal)}</p>
          <p>Discount{order.coupon ? ` (${order.coupon})` : ''}: {money(order.discount)}</p>
          <h2>Total: {money(order.total)}</h2>
        </article>
      )) : <p>No orders yet.</p>}
      <Link className="button" to="/products">Continue Shopping</Link>
    </section>
  )
}

import { NavLink, useNavigate } from 'react-router-dom'

    </nav>
  </header>
}

frontend/src/context/shopState.test.js

import test from 'node:test'
import assert from 'node:assert/strict'
import { createShopState, updateCart, couponDetails, orderTotals, completeOrder } from './shopState.js'

const customer = { name: 'Test Student', email: 'student@example.com', address: 'Test Campus' }
const date = '2026-10-19T10:00:00.000Z'
const add = (state, id) => updateCart(state, { type: 'add', id })
const monitorCart = () => add(createShopState(), 4)
const purchase = (state, coupon = '', number = 'ORD-TEST-1') => completeOrder(state, customer, coupon, number, date)

test('each demo starts with independent stock and empty history', () => {
  const first = createShopState()
  first.products[0].stock = 0
  assert.equal(createShopState().products[0].stock, 5)
  assert.deepEqual(createShopState().orders, [])
})

test('adding twice merges the cart line without reducing stock', () => {
  const state = add(monitorCart(), 4)
  assert.equal(state.cart.length, 1)
  assert.equal(state.cart[0].quantity, 2)
  assert.equal(state.products.find(item => item.id === 4).stock, 2)
})

test('out-of-stock products cannot be added', () => {
  const state = createShopState()
  assert.equal(add(state, 3), state)
})

test('repeated additions cannot exceed inventory', () => {
  const state = add(monitorCart(), 4)
  assert.equal(add(state, 4), state)
})

test('quantity updates accept the stock boundary', () => {
  const state = updateCart(monitorCart(), { type: 'quantity', id: 4, quantity: 2 })
  assert.equal(state.cart[0].quantity, 2)
})

for (const quantity of [0, -1, 1.5, 3, NaN]) {
  test(`quantity ${quantity} is rejected without changing the cart`, () => {
    const state = monitorCart()
    assert.equal(updateCart(state, { type: 'quantity', id: 4, quantity }), state)
  })
}

test('remove and clear leave inventory unchanged', () => {
  const state = monitorCart()
  for (const action of [{ type: 'remove', id: 4 }, { type: 'clear' }]) {
    const next = updateCart(state, action)
    assert.deepEqual(next.cart, [])
    assert.deepEqual(next.products, state.products)
  }
})

test('SAVE10 computes a 10 percent discount', () => {
  assert.deepEqual(orderTotals(monitorCart().cart, 'SAVE10'), {
    subtotal: 15000, discount: 1500, total: 13500, coupon: 'SAVE10',
  })
})

test('WELCOME20 computes a 20 percent discount', () => {
  assert.equal(orderTotals(monitorCart().cart, 'WELCOME20').total, 12000)
})

test('empty coupon applies no discount', () => {
  assert.equal(orderTotals(monitorCart().cart, '   ').discount, 0)
})

test('coupon normalization accepts surrounding spaces and lowercase', () => {
  assert.deepEqual(couponDetails(' save10 '), { code: 'SAVE10', rate: 0.1 })
})

test('invalid and inherited-object coupon names are rejected', () => {
  for (const code of ['INVALID', 'constructor', '__proto__']) {
    assert.throws(() => couponDetails(code), /Invalid coupon/)
  }
})

test('checkout commits inventory, immutable snapshot, date and discounted total together', () => {
  const state = monitorCart()
  const next = purchase(state, 'SAVE10')
  assert.equal(next.products.find(item => item.id === 4).stock, 1)
  assert.equal(next.products.find(item => item.id === 1).stock, 5)
  assert.deepEqual(next.cart, [])
  assert.equal(next.orders[0].number, 'ORD-TEST-1')
  assert.equal(next.orders[0].date, date)
  assert.equal(next.orders[0].total, 13500)
  assert.equal(next.orders[0].items[0].quantity, 1)
  assert.equal(state.products.find(item => item.id === 4).stock, 2)
  assert.equal(state.orders.length, 0)
  assert.notEqual(next.orders[0].items[0], state.cart[0])
})

test('second order exhausts inventory and history is newest first', () => {
  const first = purchase(monitorCart(), 'SAVE10')
  const second = purchase(add(first, 4), 'WELCOME20', 'ORD-TEST-2')
  assert.equal(second.products.find(item => item.id === 4).stock, 0)
  assert.deepEqual(second.orders.map(order => order.number), ['ORD-TEST-2', 'ORD-TEST-1'])
  assert.equal(second.orders[1].total, 13500)
  assert.equal(second.orders[0].total, 12000)
  assert.equal(add(second, 4), second)
})

test('empty-cart checkout cannot create a duplicate order', () => {
  const state = purchase(monitorCart())
  assert.throws(() => purchase(state), /cart is empty/)
  assert.equal(state.orders.length, 1)
})

test('invalid customer or coupon cannot decrement stock or append an order', () => {
  const state = monitorCart()
  for (const input of [
    { ...customer, name: '  ' }, { ...customer, email: 'invalid' }, { ...customer, address: '' },
  ]) {
    assert.throws(() => completeOrder(state, input, '', 'ORD-TEST', date))
  }
  assert.throws(() => purchase(state, 'INVALID'), /Invalid coupon/)
  assert.equal(state.products.find(item => item.id === 4).stock, 2)
  assert.equal(state.orders.length, 0)
})

test('checkout revalidates inventory before changing any state', () => {
  const state = monitorCart()
  state.products.find(item => item.id === 4).stock = 0
  assert.throws(() => purchase(state), /no longer in stock/)
  assert.equal(state.cart.length, 1)
  assert.equal(state.orders.length, 0)
})
