export class LoginPage {
  constructor(page) {
    this.page = page
    this.heading = page.getByRole('heading', { name: 'Welcome to TestLab Store', exact: true })
    this.email = page.getByRole('textbox', { name: 'Email', exact: true })
    this.password = page.getByLabel('Password', { exact: true })
    this.submitButton = page.getByRole('button', { name: 'Log in', exact: true })
  }

  async open() {
    await this.page.goto('/')
  }

  async login({ email, password }) {
    await this.email.fill(email)
    await this.password.fill(password)
    await this.submitButton.click()
  }
}

pages/ProductsPage.js

export class ProductsPage {
  constructor(page) {
    this.page = page
    this.heading = page.getByRole('heading', { name: 'Products', exact: true })
    this.searchInput = page.getByRole('searchbox', { name: 'Search products', exact: true })
    this.categorySelect = page.getByRole('combobox', { name: 'Category', exact: true })
    this.sortSelect = page.getByRole('combobox', { name: 'Sort products', exact: true })
    this.cards = page.getByRole('article')
    this.productNames = this.cards.getByRole('heading', { level: 2 })
    this.noResults = page.getByText('No products found.', { exact: true })
    const navigation = page.getByRole('navigation', { name: 'Main navigation' })
    this.productsLink = navigation.getByRole('link', { name: 'Products', exact: true })
    this.cartLink = navigation.getByRole('link', { name: /^Cart \(\d+\)$/ })
  }

  product(name) {
    return this.page.getByRole('article', { name, exact: true })
  }

  addButton(name) {
    return this.product(name).getByRole('button', { name: `Add ${name} to cart`, exact: true })
  }

  async search(value) {
    await this.searchInput.fill(value)
  }

  async filterCategory(label) {
    await this.categorySelect.selectOption({ label })
  }

  async sortBy(label) {
    await this.sortSelect.selectOption({ label })
  }

  async addToCart(name) {
    await this.addButton(name).click()
  }

  // SPA links preserve the app's in-memory cart, stock, and history.
  async open() {
    await this.productsLink.click()
  }

  async openCart() {
    await this.cartLink.click()
  }
}

export class CheckoutPage {
  constructor(page) {
    this.page = page
    this.heading = page.getByRole('heading', { name: 'Checkout', exact: true })
    this.fullName = page.getByRole('textbox', { name: 'Full Name', exact: true })
    this.email = page.getByRole('textbox', { name: 'Email', exact: true })
    this.address = page.getByRole('textbox', { name: 'Address', exact: true })
    this.couponInput = page.getByRole('textbox', { name: 'Coupon code', exact: true })
    this.applyCouponButton = page.getByRole('button', { name: 'Apply coupon', exact: true })
    this.placeOrderButton = page.getByRole('button', { name: 'Place order', exact: true })
    this.alerts = page.getByRole('alert')
    this.couponStatus = page.getByRole('status')
    this.subtotal = page.locator('p').filter({ hasText: /^Subtotal:/ }).locator('strong')
    this.discount = page.locator('p').filter({ hasText: /^Discount:/ }).locator('strong')
    this.total = page.locator('p').filter({ hasText: /^Total:/ }).locator('strong')
    // Confirmation is part of the checkout workflow; no sixth page object is needed.
    this.successMessage = page.getByRole('status').filter({ hasText: 'Your demo order was placed successfully.' })
    this.orderNumber = page.getByText(/^Order number: ORD-/)
    this.confirmationTotal = page.getByRole('heading', { name: /^Total:/ })
    this.confirmationItems = page.getByRole('listitem')
    this.historyLink = page.getByRole('link', { name: 'View order history', exact: true })
    this.continueShoppingLink = page.getByRole('link', { name: 'Continue Shopping', exact: true })
  }

  async fillCustomer({ name, email, address }) {
    await this.fullName.fill(name)
    await this.email.fill(email)
    await this.address.fill(address)
  }

  async applyCoupon(code) {
    await this.couponInput.fill(code)
    await this.applyCouponButton.click()
  }

  async submit() {
    await this.placeOrderButton.click()
  }

  async readOrderNumber() {
    const text = await this.orderNumber.innerText()
    return text.replace('Order number: ', '').trim()
  }

  async openHistory() {
    await this.historyLink.click()
  }

  async continueShopping() {
    await this.continueShoppingLink.click()
  }
}

utils/values.js

export function formatPrice(amount) {
  return new Intl.NumberFormat('en-IN', {
    style: 'currency', currency: 'INR', maximumFractionDigits: 0,
  }).format(amount)
}

export function namesByPrice(direction = 'ascending', products) {
  const multiplier = direction === 'ascending' ? 1 : -1
  return [...products].sort((a, b) => multiplier * (a.price - b.price) || a.name.localeCompare(b.name))
    .map(product => product.name)
}

Scope
JavaScript-only automation for the existing local React SUT. Application source is unchanged.

2. ARCHITECTURE

Tests express scenario expectations. Page objects own locators and user actions.
Fixtures instantiate the five page objects on Playwright's test-scoped page.
Flows compose UI actions and assert navigation readiness.
Independent test data defines expected catalog contents, rather than importing the SUT's data.

Role/label-based locators are used for interactive controls and named cards.
Small structural locators identify unnamed currency values and semantic time elements.
No fixed sleeps, XPath, positional nth() selectors, or network-idle waits are used.
Playwright locator assertions retry until the expected UI appears.

All tests use beforeEach hooks. Each test gets a fresh browser context and logs in via the UI.
Only LoginPage.open() uses page.goto('/'). Subsequent navigation uses SPA links because a
full page load resets the SUT's React state. Orders tests create their own order history.
Tests do not depend on suite order and run with two local workers (one in CI).

CheckoutPage also owns checkout confirmation locators/actions, keeping the requested five POMs.
The demo login does not authenticate real credentials; the fixture uses the documented demo values.
External Google font requests are blocked so fonts cannot slow screenshots or introduce an external
network dependency. Application requests and application state are not mocked.
